## Content

`StepList` follows the [Collection Components API](collections?component=StepList), accepting both static and dynamic collections. Use `id` on each `Step` (this maps to v3 `Item` `key`).

```tsx render
"use client";
import {StepList, Step} from 'react-aria-components/StepList';

const steps = [
  {id: 'account', name: 'Account'},
  {id: 'shipping', name: 'Shipping'},
  {id: 'payment', name: 'Payment'}
];

<StepList aria-label="Checkout" items={steps} defaultLastCompletedStep="account">
  {item => <Step id={item.id}>{item.name}</Step>}
</StepList>
```

## Selection and completion

A step is always selected. The default is the first incomplete, non-disabled step. Users can return to completed steps, and can advance only to the first incomplete step. Selecting a later step advances `lastCompletedStep` to the previous key.

```tsx render
"use client";
import type {Key} from 'react-aria-components/StepList';
import {StepList, Step} from 'react-aria-components/StepList';
import {useState} from 'react';

function Example() {
  let [selectedKey, setSelectedKey] = useState<Key>('shipping');
  let [lastCompletedStep, setLastCompletedStep] = useState<Key | null>('account');

  return (
    <StepList
      aria-label="Checkout"
      selectedKey={selectedKey}
      onSelectionChange={setSelectedKey}
      lastCompletedStep={lastCompletedStep}
      onLastCompletedStepChange={setLastCompletedStep}>
      <Step id="account">Account</Step>
      <Step id="shipping">Shipping</Step>
      <Step id="payment">Payment</Step>
    </StepList>
  );
}
```

## Styling
